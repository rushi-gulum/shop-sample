   903→   903→    const w = Math.floor(daysAgo / 7);
   904→   904→    return w === 1 ? "1 week ago" : `${w} weeks ago`;
   905→   905→  }
   906→   906→  const m = Math.floor(daysAgo / 30);
   907→   907→  return m === 1 ? "1 month ago" : `${m} months ago`;
   908→   908→}

   910→   910→/** Deterministic pseudo-random reviews for a product (same output every render/SSR). */
   911→   911→export function reviewsFor(productId: string): Review[] {
   912→   912→  const p = PRODUCT_MAP[productId];
   913→   913→  if (!p) return [];
   914→   914→  const seed = hashString(productId);
   915→   915→  const count = 4 + (seed % 3); // 4..6 reviews
   916→   916→  const rating = p.rating;
   917→   917→
   918→   918→  const reviews: { review: Review; daysAgo: number }[] = [];
   919→   919→  const usedAuthorIdx = new Set<number>();
   920→   920→  const usedContentKeys = new Set<string>();
   921→   921→  for (let i = 0; i < count; i++) {
   922→   922→    // deterministic star pick biased by the product rating
   923→   923→    const roll = ((seed >>> (i * 3 + 1)) % 100) / 100;
   924→   924→    let stars: number;
   925→   925→    if (rating >= 4.5) stars = roll < 0.68 ? 5 : roll < 0.94 ? 4 : 3;
   926→   926→    else if (rating >= 4.0) stars = roll < 0.52 ? 5 : roll < 0.9 ? 4 : 3;
   927→   927→    else if (rating >= 3.5) stars = roll < 0.34 ? 5 : roll < 0.74 ? 4 : roll < 0.94 ? 3 : 2;
   928→   928→    else stars = roll < 0.2 ? 5 : roll < 0.55 ? 4 : roll < 0.85 ? 3 : 2;
   929→   929→
   930→   930→    const pool = REVIEW_CONTENT[stars];
   931→   931→    let cIdx = (seed >>> (i * 5 + 2)) % pool.length;
   932→   932→    let guard = 0;
   933→   933→    while (usedContentKeys.has(`${stars}:${cIdx}`) && guard < pool.length * 2) {
   934→   934→      cIdx = (cIdx + 1) % pool.length;
   935→   935→      guard++;
   936→   936→    }
   937→   937→    usedContentKeys.add(`${stars}:${cIdx}`);
   938→   938→    const content = pool[cIdx];
   939→   939→
   940→   940→    let aIdx = (seed >>> (i * 4 + 3)) % REVIEW_AUTHORS.length;
   941→   941→    guard = 0;
   942→   942→    while (usedAuthorIdx.has(aIdx) && guard < REVIEW_AUTHORS.length) {
   943→   943→      aIdx = (aIdx + 1) % REVIEW_AUTHORS.length;
   944→   944→      guard++;
   945→   945→    }
   946→   946→    usedAuthorIdx.add(aIdx);
   947→   947→    const author = REVIEW_AUTHORS[aIdx];
   948→   948→
   949→   949→    const daysAgo = REVIEW_DAY_GAPS[(seed >>> (i * 3 + 5)) % REVIEW_DAY_GAPS.length];
   950→   950→
   951→   951→    reviews.push({
   952→   952→      daysAgo,
   953→   953→      review: {
   954→   954→        id: `${productId}-r${i}`,
   955→   955→        author,
   956→   956→        rating: stars,
   957→   957→        date: relativeDate(daysAgo),
   958→   958→        title: content.title,
   959→   959→        body: content.body,
   960→   960→        verified: i % 4 !== 3,
   961→   961→        helpful: 3 + ((seed >>> (i * 6 + 4)) % 46),
   962→   962→      },
   963→   963→    });
   964→   964→  }
   965→   965→  // newest first
   966→   966→  return reviews.sort((a, b) => a.daysAgo - b.daysAgo).map((r) => r.review);
   967→   967→}

   969→   969→/** Deterministic 5→1 star distribution that averages roughly to the product rating. */
   970→   970→export function ratingBreakdown(p: Product): { stars: number; pct: number; count: number }[] {
   971→   971→  const r = p.rating;
   972→   972→  const five = Math.round(Math.min(82, Math.max(25, (r - 2.2) * 34)));
   973→   973→  const one = Math.round(Math.min(18, Math.max(1, (5 - r) * 7)));
   974→   974→  const four = Math.round((100 - five - one) * 0.52);
   975→   975→  const three = Math.round((100 - five - one - four) * 0.55);
   976→   976→  const two = Math.max(0, 100 - five - four - three - one);
   977→   977→  return [five, four, three, two, one].map((pct, i) => ({
   978→   978→    stars: 5 - i,
   979→   979→    pct,
   980→   980→    count: Math.round((pct / 100) * p.ratingCount),
   981→   981→  }));
   982→   982→}
