   890→function hashString(s: string): number {
   891→  let h = 2166136261;
   892→  for (let i = 0; i < s.length; i++) {
   893→    h ^= s.charCodeAt(i);
   894→    h = Math.imul(h, 16777619);
   895→  }
   896→  return h >>> 0;
   897→}

   899→function relativeDate(daysAgo: number): string {
   900→  if (daysAgo <= 1) return "yesterday";
   901→  if (daysAgo < 7) return `${daysAgo} days ago`;
   902→  if (daysAgo < 30) {
   903→    const w = Math.floor(daysAgo / 7);
   904→    return w === 1 ? "1 week ago" : `${w} weeks ago`;
   905→  }
   906→  const m = Math.floor(daysAgo / 30);
   907→  return m === 1 ? "1 month ago" : `${m} months ago`;
   908→}

   910→/** Deterministic pseudo-random reviews for a product (same output every render/SSR). */
   911→export function reviewsFor(productId: string): Review[] {
   912→  const p = PRODUCT_MAP[productId];
   913→  if (!p) return [];
   914→  const seed = hashString(productId);
   915→  const count = 4 + (seed % 3); // 4..6 reviews
   916→  const rating = p.rating;
   917→
   918→  const reviews: { review: Review; daysAgo: number }[] = [];
   919→  const usedAuthorIdx = new Set<number>();
   920→  const usedContentKeys = new Set<string>();
   921→  for (let i = 0; i < count; i++) {
   922→    // deterministic star pick biased by the product rating
   923→    const roll = ((seed >>> (i * 3 + 1)) % 100) / 100;
   924→    let stars: number;
   925→    if (rating >= 4.5) stars = roll < 0.68 ? 5 : roll < 0.94 ? 4 : 3;
   926→    else if (rating >= 4.0) stars = roll < 0.52 ? 5 : roll < 0.9 ? 4 : 3;
   927→    else if (rating >= 3.5) stars = roll < 0.34 ? 5 : roll < 0.74 ? 4 : roll < 0.94 ? 3 : 2;
   928→    else stars = roll < 0.2 ? 5 : roll < 0.55 ? 4 : roll < 0.85 ? 3 : 2;
   929→
   930→    const pool = REVIEW_CONTENT[stars];
   931→    let cIdx = (seed >>> (i * 5 + 2)) % pool.length;
   932→    let guard = 0;
   933→    while (usedContentKeys.has(`${stars}:${cIdx}`) && guard < pool.length * 2) {
   934→      cIdx = (cIdx + 1) % pool.length;
   935→      guard++;
   936→    }
   937→    usedContentKeys.add(`${stars}:${cIdx}`);
   938→    const content = pool[cIdx];
   939→
   940→    let aIdx = (seed >>> (i * 4 + 3)) % REVIEW_AUTHORS.length;
   941→    guard = 0;
   942→    while (usedAuthorIdx.has(aIdx) && guard < REVIEW_AUTHORS.length) {
   943→      aIdx = (aIdx + 1) % REVIEW_AUTHORS.length;
   944→      guard++;
   945→    }
   946→    usedAuthorIdx.add(aIdx);
   947→    const author = REVIEW_AUTHORS[aIdx];
   948→
   949→    const daysAgo = REVIEW_DAY_GAPS[(seed >>> (i * 3 + 5)) % REVIEW_DAY_GAPS.length];
   950→
   951→    reviews.push({
   952→      daysAgo,
   953→      review: {
   954→        id: `${productId}-r${i}`,
   955→        author,
   956→        rating: stars,
   957→        date: relativeDate(daysAgo),
   958→        title: content.title,
   959→        body: content.body,
   960→        verified: i % 4 !== 3,
   961→        helpful: 3 + ((seed >>> (i * 6 + 4)) % 46),
   962→      },
   963→    });
   964→  }
   965→  // newest first
   966→  return reviews.sort((a, b) => a.daysAgo - b.daysAgo).map((r) => r.review);
   967→}

   969→/** Deterministic 5→1 star distribution that averages roughly to the product rating. */
   970→export function ratingBreakdown(p: Product): { stars: number; pct: number; count: number }[] {
   971→  const r = p.rating;
   972→  const five = Math.round(Math.min(82, Math.max(25, (r - 2.2) * 34)));
   973→  const one = Math.round(Math.min(18, Math.max(1, (5 - r) * 7)));
   974→  const four = Math.round((100 - five - one) * 0.52);
   975→  const three = Math.round((100 - five - one - four) * 0.55);
   976→  const two = Math.max(0, 100 - five - four - three - one);
   977→  return [five, four, three, two, one].map((pct, i) => ({
   978→    stars: 5 - i,
   979→    pct,
   980→    count: Math.round((pct / 100) * p.ratingCount),
   981→  }));
   982→}
